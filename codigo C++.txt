#include <iostream>

int main(int argc, char *argv[]) {
	int Contas[4][5];
	int Falhas[4];
	int i = 0, j, registros = 0;
	int conts_al = 0, qtd_falhas = 0;
	
	char continuar;
	
	std::cout << "** Simulador de Alerta de Tentativas de Login **" << std::endl;
	std::cout << "Digite 0 se falhou, senao digite 1." << std::endl << std::endl;
	
	do {
		std::cout << "Conta " << i + 1 << std::endl;
		registros = i + 1;
		j = 0;
		
		while (j < 5){
			std::cin >> Contas[i][j];
			if (Contas[i][j] < 0 || Contas[i][j] > 1)
				std::cout << "Numero invalido, digite 0 ou 1." << std::endl;
			else j++;
		}
		
		if (i + 1 < 4){
			std::cout << "Deseja continuar? (Y/N)" << std::endl;
			std::cin >> continuar;
		}
		
		i++;
	} while (i < 4 && (continuar == 'y' || continuar == 'Y'));
	
	for (i=0; i<registros; i++){
		Falhas[i] = 0;
		
		for (j=0; j<5; j++)
			if (Contas[i][j] == 0) ++Falhas[i];
		
		if (Falhas[i] >= 3) ++conts_al;
		if (Falhas[i] > qtd_falhas) qtd_falhas = Falhas[i];
	}
	
	std::cout << "Situacao para cada conta:" << std::endl;
	for (i=0; i<registros; i++){
		std::cout << "Conta " << i + 1 << ": ";
		
		for (j=0; j<5; j++) std::cout << Contas[i][j] << " ";
		std::cout << "-> " << Falhas[i] << " falhas -> ";
		
		if (Falhas[i] >= 3) std::cout << "ALERTA" << std::endl;
		else std::cout << "OK" << std::endl;
	}
	
	std::cout << std::endl << "Quantidade de contas com falha: " << conts_al << std::endl;
	std::cout << "Maior quantidade de falhas: " << qtd_falhas;
	
	return 0;
}
